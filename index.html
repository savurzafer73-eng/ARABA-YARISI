using UnityEngine; public class CarController : MonoBehaviour { [Header("Tekerlek Konfigürasyonu (Wheel Colliders)")] public WheelCollider solOnWheel; public WheelCollider sagOnWheel; public WheelCollider solArkaWheel; public WheelCollider sagArkaWheel; [Header("Tekerlek Modelleri (Mesh Transforms)")] public Transform solOnTransform; public Transform sagOnTransform; public Transform solArkaTransform; public Transform sagArkaTransform; [Header("Araba Ayarları")] public float motorGucu = 1500f; public float frenGucu = 3000f; public float maxDonusAcisi = 30f; public float elFreniGucu = 5000f; public Vector3 agirlikMerkeziOffset = new Vector3(0, -0.5f, 0); // Giriş Değişkenleri (Dokunmatik Butonlardan Tetiklenir) private float dikeyGiris = 0f; // 1 = Gaz, -1 = Fren/Geri, 0 = Boşta private float yatayGiris = 0f; // 1 = Sağ, -1 = Sol, 0 = Düz private bool elFreniCekili = false; private Rigidbody rb; void Start() { rb = GetComponent(); // Arabanın takla atmasını önlemek için ağırlık merkezini aşağı çekiyoruz if (rb != null) { rb.centerOfMass += agirlikMerkeziOffset; } } void FixedUpdate() { MotoruYonet(); DireksiyonuYonet(); TekerlekleriGuncelle(); } // --- SÜRÜŞ VE FİZİK MANTIĞI --- private void MotoruYonet() { if (elFreniCekili) { // El freni çekildiyse arka tekerleri kilitle solArkaWheel.brakeTorque = elFreniGucu; sagArkaWheel.brakeTorque = elFreniGucu; solOnWheel.motorTorque = 0; sagOnWheel.motorTorque = 0; } else { // Normal Sürüş (Arkadan İtişli - RWD) solArkaWheel.motorTorque = dikeyGiris * motorGucu; sagArkaWheel.motorTorque = dikeyGiris * motorGucu; // Fren Mantığı float fren = (dikeyGiris == 0) ? frenGucu * 0.1f : 0f; solOnWheel.brakeTorque = fren; sagOnWheel.brakeTorque = fren; solArkaWheel.brakeTorque = fren; sagArkaWheel.brakeTorque = fren; } } private void DireksiyonuYonet() { float donusAcisi = yatayGiris * maxDonusAcisi; solOnWheel.steerAngle = donusAcisi; sagOnWheel.steerAngle = donusAcisi; } private void TekerlekleriGuncelle() { TekerlegiGorselleStil(solOnWheel, solOnTransform); TekerlegiGorselleStil(sagOnWheel, sagOnTransform); TekerlegiGorselleStil(solArkaWheel, solArkaTransform); TekerlegiGorselleStil(sagArkaWheel, sagArkaTransform); } private void TekerlegiGorselleStil(WheelCollider collider, Transform trans) { if (trans == null) return; Vector3 pos; Quaternion rot; collider.GetWorldPose(out pos, out rot); trans.position = pos; trans.rotation = rot; } // --- DOKUNMATİK BUTON HAREKET FONKSİYONLARI --- public void GazBas() { dikeyGiris = 1f; } public void FrenBas() { dikeyGiris = -1f; } public void DikeyBrak() { dikeyGiris = 0f; } public void SolDon() { yatayGiris = -1f; } public void SagDon() { yatayGiris = 1f; } public void YatayBrak() { yatayGiris = 0f; } public void ElFreniBas() { elFreniCekili = true; } public void ElFreniBrak() { elFreniCekili = false; } }
